# Server-side prefs implementation plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Make `/app/customize` real: visual prefs persist across devices via DDB, and two of the three Behavior toggles (Send on Enter, Suggest follow-ups) actually affect runtime behavior.

**Architecture:** New `PK=USER#{u}, SK=PREFS` JSON-blob row. `GET/PUT /api/me/prefs`. `useChannelPrefs` adds a one-shot server hydrate + debounced write-through; localStorage stays canonical for offline. Follow-ups generated by a Haiku one-shot after `done` SSE event, emitted as a new `follow_ups_suggested` event the SPA renders as a chip row under the last assistant turn.

**Tech Stack:** Python 3 / FastAPI / boto3 / pydantic v2 / Strands Agents / React 19 / Vite / vitest / pytest.

**Spec:** `docs/superpowers/specs/2026-06-02-server-side-prefs-design.md` — read it first.

```python
def get_prefs(user_id: str) -> Prefs:
    """Return the user's prefs, or default Prefs if no row exists."""
    result = _get_table().get_item(
        Key={"PK": f"USER#{user_id}", "SK": "PREFS"}
    )
    item = result.get("Item")
    if not item:
        return Prefs()
    return Prefs(**(item.get("prefs") or {}))


def put_prefs(user_id: str, updates: dict[str, Any]) -> Prefs:
    """Merge `updates` into the user's prefs row and return the merged Prefs.

    Validates the merged result against Prefs (so unknown keys raise).
    Initialises with defaults when no row exists.
    """
    current = get_prefs(user_id).model_dump()
    current.update(updates)
    merged = Prefs(**current)  # raises ValidationError on unknown / bad type
    _get_table().put_item(
        Item={
            "PK": f"USER#{user_id}",
            "SK": "PREFS",
            "prefs": merged.model_dump(),
        }
    )
    return merged
```

```python
"""User preferences API — GET / PUT /api/me/prefs."""
from __future__ import annotations

import pydantic
from fastapi import APIRouter, Depends, HTTPException
from pydantic import BaseModel

from .. import storage
from ..models import Prefs
from ._auth import require_mgmt_user

router = APIRouter()


class PrefsEnvelope(BaseModel):
    prefs: Prefs


class PrefsUpdateRequest(BaseModel):
    prefs: dict


@router.get("/api/me/prefs", response_model=PrefsEnvelope)
def get_my_prefs(user=Depends(require_mgmt_user)) -> PrefsEnvelope:
    return PrefsEnvelope(prefs=storage.get_prefs(user.sub))


@router.put("/api/me/prefs", status_code=204)
def put_my_prefs(
    body: PrefsUpdateRequest, user=Depends(require_mgmt_user)
) -> None:
    try:
        storage.put_prefs(user.sub, body.prefs)
    except pydantic.ValidationError as exc:
        # Surface as 422 with the offending key(s)
        raise HTTPException(status_code=422, detail=exc.errors())
```

Add server-sync wiring at module level:

```javascript
// Server key (snake_case) ↔ hook key (camelCase) mapping.
const SERVER_KEY = {
  theme: "theme", accent: "accent", density: "density", shape: "shape",
  font: "font", model: "model", effort: "effort",
  sendOnEnter: "send_on_enter",
  showReasoning: "show_reasoning",
  suggestFollowups: "suggest_followups",
};
const HOOK_KEY = Object.fromEntries(
  Object.entries(SERVER_KEY).map(([h, s]) => [s, h])
);

let hydrated = false;
let hydratePromise = null;

function shouldHydrate() {
  try { return Boolean(localStorage.getItem("starter_mgmt_token")); }
  catch { return false; }
}

async function hydrateFromServer() {
  if (hydrated) return;
  if (hydratePromise) return hydratePromise;
  hydratePromise = (async () => {
    try {
      const { getPrefs } = await import("../api.js");
      const serverPrefs = await getPrefs();
      const next = { ...snapshot };
      for (const [serverKey, val] of Object.entries(serverPrefs)) {
        const hookKey = HOOK_KEY[serverKey];
        if (!hookKey) continue;
        const stringVal = typeof val === "boolean" ? (val ? "1" : "0") : String(val);
        next[hookKey] = stringVal;
        writePref(STORAGE_KEYS[hookKey], stringVal);
      }
      snapshot = Object.freeze(next);
      listeners.forEach((l) => l());
    } catch {
      /* localStorage stays canonical; degrade silently */
    } finally {
      hydrated = true;
    }
  })();
  return hydratePromise;
}

// Debounced server PUT, per-key
const pendingPuts = {};
function schedulePutPref(name, value) {
  if (!shouldHydrate()) return;
  clearTimeout(pendingPuts[name]);
  pendingPuts[name] = setTimeout(async () => {
    delete pendingPuts[name];
    try {
      const { putPrefs } = await import("../api.js");
      const serverKey = SERVER_KEY[name];
      const serverVal = ["sendOnEnter", "showReasoning", "suggestFollowups"].includes(name)
        ? value === "1"
        : value;
      await putPrefs({ [serverKey]: serverVal });
    } catch {
      /* swallow; localStorage already updated */
    }
  }, 200);
}
```

Update `setPref` to call `schedulePutPref(name, value)` after `writePref`.
